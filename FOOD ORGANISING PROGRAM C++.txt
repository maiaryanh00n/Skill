
#include <iostream>
using namespace std;

int main() {
    int choice, qty;
    int total = 0;

    cout << "----- FOOD MENU -----" << endl;
    cout << "1. Pizza  - Rs.200" << endl;
    cout << "2. Burger - Rs.100" << endl;
    cout << "3. Sandwich - Rs.80" << endl;
    cout << "4. Exit" << endl;

    do {
        cout << "\nEnter your choice: ";
        cin >> choice;

        if (choice >= 1 && choice <= 3) {
            cout << "Enter quantity: ";
            cin >> qty;

            if (qty <= 0) {
                cout << "Invalid quantity!" << endl;
                continue;
            }

            if (choice == 1)
                total += 200 * qty;
            else if (choice == 2)
                total += 100 * qty;
            else
                total += 80 * qty;

            cout << "Order added successfully!" << endl;
        }
        else if (choice != 4) {
            cout << "Invalid choice!" << endl;
        }

    } while (choice != 4);

    cout << "\nTotal Bill = Rs." << total << endl;
    cout << "Thank you for ordering!" << endl;

    return 0;
}
