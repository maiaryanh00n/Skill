#include <iostream>
#include <sstream>
using namespace std;

int main() {
    string s, word, largest;
    cout << "Enter a string: ";
    getline(cin, s);

    stringstream ss(s);

    while (ss >> word) {
        if (word.length() > largest.length())
            largest = word;
    }

    cout << "Largest word: " << largest;
    return 0;
}
