#include <iostream>
#include <sstream>
using namespace std;

int main() {
    string s, word, small;
    getline(cin, s);
    stringstream ss(s);

    while (ss >> word) {
        if (small.empty() || word.length() < small.length())
            small = word;
    }

    cout << "Shortest word: " << small;
    return 0;
}
