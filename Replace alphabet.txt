#include <iostream>
using namespace std;

int main() {
    string s;
    cout << "Enter a string: ";
    cin >> s;

    for (int i = 0; s[i] != '\0'; i++) {
        if (s[i] >= 'a' && s[i] <= 'z')
            s[i] = (s[i] == 'z') ? 'a' : s[i] + 1;
        else if (s[i] >= 'A' && s[i] <= 'Z')
            s[i] = (s[i] == 'Z') ? 'A' : s[i] + 1;
    }

    cout << "New string: " << s;
    return 0;
}
